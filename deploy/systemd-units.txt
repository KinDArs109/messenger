=== messenger.service ===
[Unit]
Description=Мессенджер
After=network-online.target postgresql.service
Wants=network-online.target
Requires=postgresql.service

[Service]
Type=simple
User=messenger
Group=messenger
WorkingDirectory=/opt/messenger/apps/server
Environment=HOME=/opt/messenger
# Собранный сервер, а не исходники через tsx: он не переваривает
# TypeScript при каждом старте, поднимается вдвое быстрее (1,6 с
# против 2,8) и занимает меньше памяти. Карты исходников включены,
# чтобы в журнале падений стояли настоящие имена файлов и строки.
ExecStart=/usr/bin/node --enable-source-maps --env-file=.env dist/index.js
Restart=always
RestartSec=3

# Процесс смотрит в интернет, поэтому прав у него ровно столько,
# сколько нужно: своя папка на запись, остальная система только
# на чтение, повышение прав запрещено.
NoNewPrivileges=true
PrivateTmp=true
ProtectSystem=strict
ProtectHome=true
ReadWritePaths=/opt/messenger

[Install]
WantedBy=multi-user.target

=== messenger-backup.service ===
[Unit]
Description=Копия мессенджера: база и загруженные файлы
After=postgresql.service

[Service]
Type=oneshot
ExecStart=/opt/messenger/backup.sh

[Timer]

[Install]

[Install]
